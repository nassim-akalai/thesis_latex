% ------------------------------------------------------------
% Eine Seite pro Folie mit hellgrauem Rahmen
% ------------------------------------------------------------
\newcommand{\SlidePage}[1]{%
  \thispagestyle{fancy}%
  \noindent
  \begin{minipage}[t][\SlideAreaHeight][c]{\textwidth}
    \centering
    \setlength{\fboxrule}{0.6pt}% Dicke der Rahmenlinie
    \setlength{\fboxsep}{0pt}%   Bündiger Abschluss ohne Innenabstand
    \fcolorbox{SlideBorderGray}{white}{%
      \includegraphics[
        page=#1,
        width=\dimexpr\textwidth - 2\fboxrule\relax,
        height=\dimexpr\SlideAreaHeight - 2\fboxrule\relax,
        keepaspectratio
      ]{\PresentationFile}%
    }%
  \end{minipage}%
  \HandoutRule
  \noindent\textbf{Ergänzungen zu Folie #1}\par
  \vspace{0.12cm}%
  \begin{minipage}[t][\TextAreaHeight][t]{\textwidth}
    \setlength{\parindent}{0pt}%
    \setlength{\parskip}{0.55em}%
    \raggedright
    \PrintSlideText{#1}%
  \end{minipage}%
  \clearpage
}

\newcommand{\AllSlides}{%
  \DetermineSlideCount
  \CurrentSlide=1
  \loop
    \SlidePage{\the\CurrentSlide}%
    \advance\CurrentSlide by 1
  \ifnum\CurrentSlide<\numexpr\SlideCount+1\relax
  \repeat
}
