% ------------------------------------------------------------
% Folientexte verwalten
% ------------------------------------------------------------
% Beispiel:
% \AddSlideText{1}{In diesem Abschnitt wird ...}
% \AddSlideText{2}{Die verwendete Architektur besteht aus ...}
%
% Wird fuer eine Folie kein Text hinterlegt, erscheint eine unaufdringliche
% Markierung. Diese kann spaeter geloescht oder ersetzt werden.
\newcommand{\AddSlideText}[2]{%
  \expandafter\def\csname SlideText@#1\endcsname{#2}%
}

\newcommand{\PrintSlideText}[1]{%
  \ifcsname SlideText@#1\endcsname
    \csname SlideText@#1\endcsname
  \else
    \textcolor{NoteGray}{\textit{Ergaenzungen zu Folie #1 hier eintragen.}}%
  \fi
}

% ------------------------------------------------------------
% Beispiel-Eintraege: durch die eigenen Ergaenzungen ersetzen.
% Die Texte sollen den Folieninhalt erklaeren und nicht lediglich wiederholen.
% ------------------------------------------------------------
\AddSlideText{1}{%
  Diese Folie fuehrt in das Thema der Praxisphase ein. Hier sollte kurz
  erlaeutert werden, welches praktische Problem untersucht oder geloest wurde
  und welchen Beitrag die eigene Arbeit dazu geleistet hat.
}

% Folienzahl automatisch aus dem Praesentations-PDF bestimmen

\newcount\SlideCount
\newcount\CurrentSlide
\newcommand{\DetermineSlideCount}{%
  \pdfximage{\PresentationFile}%
  \SlideCount=\pdflastximagepages
}

% ------------------------------------------------------------
% Eine Seite pro Folie
% ------------------------------------------------------------
\newcommand{\SlidePage}[1]{%
  \thispagestyle{fancy}%
  \noindent
  \begin{minipage}[t][\SlideAreaHeight][c]{\textwidth}
    \centering
    \includegraphics[
      page=#1,
      width=\textwidth,
      height=\SlideAreaHeight,
      keepaspectratio
    ]{\PresentationFile}%
  \end{minipage}%
  \HandoutRule
  \noindent\textbf{Ergaenzungen zu Folie #1}\par
  \vspace{0.12cm}%
  \begin{minipage}[t][\TextAreaHeight][t]{\textwidth}
    \setlength{\parindent}{0pt}%
    \setlength{\parskip}{0.55em}%
    \raggedright
    \PrintSlideText{#1}%
  \end{minipage}%
  \clearpage
}

\newcommand{\AllSlides}{%
  \DetermineSlideCount
  \CurrentSlide=1
  \loop
    \SlidePage{\the\CurrentSlide}%
    \advance\CurrentSlide by 1
  \ifnum\CurrentSlide<\numexpr\SlideCount+1\relax
  \repeat
}
